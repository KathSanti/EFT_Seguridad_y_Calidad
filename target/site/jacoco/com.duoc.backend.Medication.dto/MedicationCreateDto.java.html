public record MedicationCreateDto(
    @NotBlank(message = "El nombre del medicamento es obligatorio")
    String name,

    @NotNull(message = "El costo es obligatorio")
    @PositiveOrZero(message = "El costo no puede ser un valor negativo")
    Double cost
) {}
